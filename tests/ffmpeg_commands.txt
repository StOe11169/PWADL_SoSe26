Commands to create dummy videos with ffmpeg

001 — navy background, yellow square moving horizontally, 440 Hz
ffmpeg -y `
    -f lavfi -i "color=c=navy:s=640x480:r=25:d=3" `
    -f lavfi -i "color=c=yellow:s=96x96:r=25:d=3" `
    -f lavfi -i "sine=frequency=440:sample_rate=16000:duration=3" `
    -filter_complex "[0:v][1:v]overlay=x='20+mod(t*160,W-w-40)':y='(H-h)/2':eval=frame[v]" `
    -map "[v]" -map 2:a `
    -c:v libx264 -crf 28 -pix_fmt yuv420p `
    -c:a aac -b:a 64k -ar 16000 `
    -shortest `
    tests/001-driver-yawning.mp4

002 — green background, orange rectangle moving diagonally, 660 Hz
ffmpeg -y `
  -f lavfi -i "color=c=green:s=640x480:r=25:d=3" `
  -f lavfi -i "color=c=orange:s=160x60:r=25:d=3" `
  -f lavfi -i "sine=frequency=660:sample_rate=16000:duration=3" `
  -filter_complex "[0:v][1:v]overlay=x='20+mod(t*140,W-w-40)':y='20+mod(t*100,H-h-40)':eval=frame[v]" `
  -map "[v]" -map 2:a `
  -c:v libx264 -crf 28 -pix_fmt yuv420p `
  -c:a aac -b:a 64k -ar 16000 `
  -shortest `
  tests/002-driver-normal.mp4
